string Check(ValidationReleasePreparationStatus expected, ValidationReleasePreparationStatus actual,
    ValidationReleaseMaterial? material, byte[]? preimage = null, byte[]? digest = null)
{
    string outcome = $"expected={expected} actual={actual} material={(material is null ? "null" : "present")}";
    if (expected != actual)
        throw new InvalidOperationException(outcome);
    if (expected != StructuralMatch)
    {
        Require(material is null, outcome);
        return outcome;
    }
    Require(material is not null, outcome);
    var actualPreimage = material!.Preimage.AsSpan();
    var actualDigest = material.CorrelationSha256.AsSpan();
    Require(actualPreimage.Length == 258 && actualPreimage.SequenceEqual(preimage),
        outcome + $" preimage={Convert.ToHexString(actualPreimage)}");
    Require(actualDigest.Length == 32 && actualDigest.SequenceEqual(digest),
        outcome + $" digest={Convert.ToHexString(actualDigest)}");
    return outcome + $" preimage={Convert.ToHexString(actualPreimage)} digest={Convert.ToHexString(actualDigest)}";
}

    }
    catch (Exception ex)
    {
        failed++;
        Console.WriteLine(JsonSerializer.Serialize(new { id, result = "fail", observation = ex.ToString() }));
    }
}

Local("Ownership.IndependentBackingAndStableReferences", () =>
{
    var left = Input(initial);
    var right = Input(initial);
    var first = ValidationReleasePreparation.Prepare(left, right, out var one);
    var second = ValidationReleasePreparation.Compare(left, right, expectedDigest, out var two);
    var alternative = alternate.GetProperty("projection");
    var third = ValidationReleasePreparation.Prepare(Input(alternative), Input(alternative), out var three);
    Check(StructuralMatch, first, one, witness, expectedDigest);
    Check(StructuralMatch, second, two, witness, expectedDigest);
    Check(StructuralMatch, third, three, alternateWitness, alternateDigest);
    Require(!ReferenceEquals(one, two), "same result object");
    Require(!one!.Preimage.Equals(two!.Preimage), "preimages share backing");
    Require(!one.CorrelationSha256.Equals(two.CorrelationSha256), "digests share backing");
    ImmutableArray<byte> alias = one.Preimage;
    ImmutableArray<byte> aliasDigest = one.CorrelationSha256;
    var preimageCopy = alias.ToArray();
    var digestCopy = aliasDigest.ToArray();
    Array.Fill(preimageCopy, (byte)0xD3);
    Array.Fill(digestCopy, (byte)0x96);
    Require(alias.AsSpan().SequenceEqual(witness) && aliasDigest.AsSpan().SequenceEqual(expectedDigest),
        "ordinary copies changed the original");
    Check(StructuralMatch, first, one, witness, expectedDigest);
    return Check(StructuralMatch, second, two, witness, expectedDigest);
});

static int CountClones(MethodInfo method)
{
    var codeMap = typeof(OpCodes).GetFields(BindingFlags.Public | BindingFlags.Static)
        .Where(f => f.FieldType == typeof(OpCode)).Select(f => (OpCode)f.GetValue(null)!)
        .ToDictionary(op => unchecked((ushort)op.Value));
    var il = method.GetMethodBody()!.GetILAsByteArray()!;
    var count = 0;
    for (int i = 0; i < il.Length;)
    {
        ushort key = il[i++];
        if (key == 0xFE)
            key = (ushort)(0xFE00 | il[i++]);
        var op = codeMap[key];
        if (op.OperandType == OperandType.InlineMethod)
        {
            var target = method.Module.ResolveMethod(BitConverter.ToInt32(il, i));
            if (target?.DeclaringType == typeof(Array) && target.Name == "Clone")
                count++;
        }
        i += op.OperandType switch
        {
            OperandType.InlineNone => 0,
            OperandType.ShortInlineBrTarget or OperandType.ShortInlineI or OperandType.ShortInlineVar => 1,
            OperandType.InlineVar => 2,
            OperandType.InlineBrTarget or OperandType.InlineField or OperandType.InlineI or OperandType.InlineMethod
                or OperandType.InlineSig or OperandType.InlineString or OperandType.InlineTok or OperandType.InlineType
                or OperandType.ShortInlineR => 4,
            OperandType.InlineI8 or OperandType.InlineR => 8,
            OperandType.InlineSwitch => 4 * (1 + BitConverter.ToInt32(il, i)),
            _ => throw new InvalidOperationException(op.OperandType.ToString())
        };
    }
    return count;
}

static void Require(bool condition, string detail)
{
    if (!condition)
        throw new InvalidOperationException(detail);
}

static void Properties(Type type, (string Name, Type Type)[] expected)
{
    const BindingFlags flags = BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static |
        BindingFlags.Instance | BindingFlags.DeclaredOnly;
    var all = type.GetProperties(flags);
    Require(all.Length == expected.Length, $"{type}: property count");
    foreach ((string name, Type fieldType) in expected)
    {
        var property = all.SingleOrDefault(p => p.Name == name);
        Require(property is not null && property.PropertyType == fieldType &&
            property.GetMethod?.IsAssembly == true && !property.GetMethod.IsStatic &&
            property.SetMethod is null, $"{type}.{name}: property shape");
    }
}

static void EnumShape<T>(int first, string names) where T : struct, Enum
{
    Require(Enum.GetUnderlyingType(typeof(T)) == typeof(byte) &&
        string.Join(' ', Enum.GetNames<T>()) == names &&
        Enum.GetValues<T>().Select(v => Convert.ToInt32(v)).SequenceEqual(Enumerable.Range(first, names.Count(ch => ch == ' ') + 1)),
        typeof(T).ToString());
}

static string Digest(byte[] raw) => Convert.ToHexStringLower(SHA256.HashData(raw));
static byte[]? Hex(JsonElement element, string key) =>
    element.GetProperty(key).GetString() is string text ? Convert.FromHexString(text) : null;
static byte[][] Buffers(ValidationReleaseProjection p) =>
    [p.SealedLaunchRecordHash!, p.CatalogHash!, p.ScriptHash!, p.ReleaseActionHash!, p.PreparationManifestDigest!];

static ValidationReleaseProjection? Input(JsonElement obj, ValidationReleaseProjection? sameBuffers = null,
    ValidationReleaseStep[]? sameSteps = null)
{
    if (obj.ValueKind == JsonValueKind.Null)
        return null;
    uint? U(string name) => obj.GetProperty(name) is { ValueKind: JsonValueKind.Number } value ? value.GetUInt32() : null;
    bool? B(string name) => obj.GetProperty(name) is { ValueKind: JsonValueKind.Null } ? null : obj.GetProperty(name).GetBoolean();
    Guid? G(string name) => obj.GetProperty(name).GetString() is string text ? Guid.ParseExact(text, "D") : null;
    var steps = obj.GetProperty("Steps");
    var template = sameSteps ?? (steps.ValueKind == JsonValueKind.Null ? null : steps.EnumerateArray()
        .Select(pair => new ValidationReleaseStep(pair[0].GetUInt32(), (ValidationReleaseOpcode)pair[1].GetByte())).ToArray());
    var bytes = sameBuffers is null
        ? new[] { Hex(obj,"SealedLaunch"), Hex(obj,"Catalog"), Hex(obj,"Script"), Hex(obj,"ReleaseAction"), Hex(obj,"Manifest") }
        : Buffers(sameBuffers);
    return new ValidationReleaseProjection(U("Version"), (ValidationReleasePhase?)U("Phase"),
        (ValidationReleaseTopology?)U("Topology"), obj.GetProperty("Case").GetString(),
        B("CatalogFlag"), B("FixtureFlag"), template, U("Index"), bytes[0], bytes[1], bytes[2], bytes[3],
        G("Session"), G("Execution"), G("Lease"), G("Preparation"), bytes[4]);
}
