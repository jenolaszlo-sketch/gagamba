def git(*args, data=None):
    return subprocess.run(
        ["git", "-c", "core.longpaths=true", *args], cwd=repo, input=data,
        stdout=subprocess.PIPE, check=True,
    ).stdout

def line(*args):
    return git(*args).decode().strip()

def check(condition, detail):
    if not condition:
        raise AssertionError(detail)

def sha(data):
    return hashlib.sha256(data).hexdigest()

def path_bytes(path):
    return Path(path).read_bytes()

def disk(path):
    return (repo / path).read_bytes()

def batch(rev, paths):
    paths = list(dict.fromkeys(paths))
    data = b"".join((rev + ":" + path).encode() + b"\n" for path in paths)
    raw = git("cat-file", "--batch", data=data)
    at = 0
    result = {}
    for path in paths:
        end = raw.index(b"\n", at)
        fields = raw[at:end].decode().split()
        check(len(fields) == 3 and fields[1] == "blob", (rev, path, fields))
        oid, _, size = fields
        at = end + 1
        end = at + int(size)
        result[path] = (oid, raw[at:end])
        check(raw[end:end+1] == b"\n", (rev, path))
        at = end + 1
    check(at == len(raw), rev)
    return result

def blob(rev, path):
    return batch(rev, [path])[path][1]

def json_at(rev, path):
    return json.loads(blob(rev, path))

def annex(report, file):
    return report + "-evidence/" + file

def diff(old, new):
    return set(line("diff", "--name-status", old, new).splitlines())

def manifest_rows(path, content):
    entries = {}
    for row in content.decode("utf-8-sig").splitlines():
        match = re.fullmatch(r"([0-9A-Fa-f]{64}) [ *](.+)", row)
        check(match, (path, row))
        digest, rel = match.groups()
        target = posixpath.normpath(posixpath.join(posixpath.dirname(path), rel))
        check(not target.startswith(("../", "/")) and target not in (".", "..") and target not in entries, (path, target))
        entries[target] = digest.lower()
    return entries

def check_note(path, old, new):
    first, sep, tail = old.partition(b"\n\n")
    check(sep and new.startswith(first + sep) and new.endswith(tail), path)
    note = new[len(first + sep):-len(tail)]
    check(note.startswith(b"> **ALT-B SLICE E-6") and note.endswith(b"\n\n") and
          note.count((authorization.removeprefix("docs/") + ".md").encode()) == 1, path)
    check(new == blob(auth, path), (path, "authorized byte identity"))
    return {"original": sha(old), "current": sha(new), "insertion": sha(note)}
